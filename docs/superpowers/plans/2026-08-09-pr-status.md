# PR Status Command Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `bb pr status <pr-id>` that reports Bitbucket PR merge readiness (`READY` / `NOT READY` / `UNKNOWN`) with human and JSON output.

**Architecture:** Thin API methods fetch PR data; `cmd/pr_status.go` orchestrates and builds a snapshot; `internal/status` owns pure `Evaluate` + `FormatHuman` / `FormatJSON` (no network). Soft-fail optional endpoints and branch-restriction auth into `UNKNOWN` checks.

**Tech Stack:** Go 1.25, Cobra, Bitbucket Cloud REST API 2.0, `testing` package

**Spec:** `docs/superpowers/specs/2026-06-24-pr-status-design.md`

package api

import (
	"errors"
	"fmt"
)

type HTTPError struct {
	StatusCode int
	Message    string
}

func (e *HTTPError) Error() string {
	return fmt.Sprintf("API error (%d): %s", e.StatusCode, e.Message)
}

func IsForbidden(err error) bool {
	var he *HTTPError
	return errors.As(err, &he) && he.StatusCode == 403
}

func IsUnauthorized(err error) bool {
	var he *HTTPError
	return errors.As(err, &he) && he.StatusCode == 401
}
```

In `client.go` `do()`, on `resp.StatusCode >= 400`, return `&HTTPError{StatusCode: resp.StatusCode, Message: ...}` (keep existing message extraction from `models.APIError` / raw body).

const (
	ReadinessReady    = "READY"
	ReadinessNotReady = "NOT_READY"
	ReadinessUnknown  = "UNKNOWN"

	CheckFail    = "FAIL"
	CheckUnknown = "UNKNOWN"

type ThresholdState int

const (
	ThresholdStateNone ThresholdState = iota // no requirement for this category
	ThresholdStateValue                        // numeric requirement
	ThresholdStateUnknown                      // could not evaluate requirement
)

type Threshold struct {
	State ThresholdState
	Value int
}

func ThresholdNone() Threshold    { return Threshold{State: ThresholdStateNone} }
func ThresholdValue(v int) Threshold {
	return Threshold{State: ThresholdStateValue, Value: v}
}
func ThresholdUnknown() Threshold { return Threshold{State: ThresholdStateUnknown} }

type Snapshot struct {

	State                   string
	Draft                   bool
	DestinationBranch       string

	TasksAvailable          bool
	BuildsAvailable         bool
	ConflictsAvailable      bool

	IgnoreBuilds            bool

- [ ] **Step 3: Implement `Evaluate`**

Check order (human output order):
1. `state` — OK if `OPEN`, else FAIL
2. `draft` — OK if not draft, else FAIL
3. `approvals` — compare counts vs threshold (Unknown → UNKNOWN check; None → INFO with observed only; Value → OK/FAIL)
4. `default_reviewer_approvals` — same; omit INFO line if threshold None and observed 0 optional — still include when Value/Unknown
5. `comments` — unless IgnoreComments: FAIL if unresolved > 0; if !CommentsAvailable → UNKNOWN; if ignored → INFO
6. `tasks` — same pattern with OpenTasks / IgnoreTasks / TasksAvailable
7. `builds` — threshold None → INFO (observed successful); Unknown → UNKNOWN; Value → OK/FAIL vs SuccessfulBuilds; !BuildsAvailable and Value/Unknown → UNKNOWN
8. `conflicts` — !ConflictsAvailable → UNKNOWN; else OK if 0 else FAIL

Top-level readiness:
- any FAIL → `NOT_READY`
- else any UNKNOWN among required checks → `UNKNOWN`
- else `READY`

Populate `Missing` from FAIL messages. Copy `snap.Warnings` into `Result.Warnings`. Fill `Result.PullRequest` from snapshot.

func runPRStatus(cmd *cobra.Command, args []string) error {
	prID, err := strconv.Atoi(args[0])
	// resolve workspace/repo, load config, NewClient — same as pr_view

	pr, err := client.GetPullRequest(workspace, repo, prID)
	if err != nil {
		return fmt.Errorf("failed to get pull request: %w", err)
	}

	var warnings []string
	const pageLimit = 100

	comments, err := client.ListPullRequestComments(workspace, repo, prID, pageLimit)
	commentsOK := err == nil
	if err != nil {
		warnings = append(warnings, fmt.Sprintf("comments unavailable: %v", err))
	}

	tasks, err := client.ListPullRequestTasks(workspace, repo, prID, pageLimit)
	tasksOK := err == nil
	if err != nil {
		warnings = append(warnings, fmt.Sprintf("tasks unavailable: %v", err))
	}

	statuses, err := client.ListPullRequestStatuses(workspace, repo, prID, pageLimit)
	buildsOK := err == nil
	if err != nil {
		warnings = append(warnings, fmt.Sprintf("builds unavailable: %v", err))
	}

	conflicts, err := client.ListPullRequestConflicts(workspace, repo, prID)
	conflictsOK := err == nil
	if err != nil {
		warnings = append(warnings, fmt.Sprintf("conflicts unavailable: %v", err))
	}

	rules, err := client.ListBranchRestrictions(workspace, repo, pageLimit)
	rulesOK := err == nil
	if err != nil {
		if api.IsForbidden(err) || api.IsUnauthorized(err) {
			warnings = append(warnings, "branch rules unavailable: insufficient permissions")
		} else {
			warnings = append(warnings, fmt.Sprintf("branch rules unavailable: %v", err))
		}
	}

	defaultReviewers, err := client.ListDefaultReviewers(workspace, repo, pageLimit)
	if err != nil && rulesOK {
		warnings = append(warnings, fmt.Sprintf("default reviewers unavailable: %v", err))
		defaultReviewers = nil
	}

	snap := buildSnapshot(pr, comments, commentsOK, tasks, tasksOK, statuses, buildsOK, conflicts, conflictsOK, defaultReviewers, warnings)
	overrides := buildOverrides(rulesOK)
	req := status.InferRequirements(rules, pr.Destination.Branch.Name, overrides)
	result := status.Evaluate(snap, req)

	if statusJSON {
		b, err := status.FormatJSON(result)
		if err != nil {
			return err
		}
		fmt.Println(string(b))
	} else {
		fmt.Print(status.FormatHuman(result))
	}

	if failOnNotReady && result.Readiness == status.ReadinessNotReady {
		return fmt.Errorf("pull request is not ready to merge")
	}
	return nil
}
```
